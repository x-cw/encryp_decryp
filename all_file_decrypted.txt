def search_in_sheet(sheet, startrow = 0, startcol = 0, key = 'offset_addr') :
    row = startrow
    col = startcol
    flag = 0
    rownum = 0
    colnum = 0
   # print(sheet.ncols)
   # print(sheet.nrows)
    #debug
    #print ('DEBUG Info: nrows,ncols = ', sheet.nrows, sheet.ncols) 
    for row in range(startrow, sheet.nrows) :
        for col in range(startcol, sheet.ncols) :
            if ((sheet.cell_type(row, col) == 1) and (key in sheet.cell(row, col).value)) :
                rownum = row
                colnum = col
                flag = 1
                break
        else : #if go through all columns, continue on next row.
            continue
        break        
    else : #go through all rows, can't find the expected string.
        flag = 0 
        rownum = startrow
        colnum = startcol        
    return flag, rownum, colnum 

def has_duplicates(lst):
    return len(lst) != len(set(lst))

def power_state_func(sheet):
    supply_set_list = []
    power_state_list = []
    supply_power_list = []
    supply_ground_list = []
    power_state_block = ''
    flag_sec,row_sec,col_sec = search_in_sheet(sheet,0,0,'supply-set')
    flag_init,row_init,col_init = search_in_sheet(sheet,0,0,'power-state-group')
    row_last = sheet.nrows
    col_last = sheet.ncols
    matrix = [[''] * (col_last-col_init) for _ in range(row_sec-row_init)]
    if(flag_sec and flag_init) :
        for row in range(row_sec+1,row_last):
            supply_set_list.append(sheet.cell(row,col_sec).value.strip())
            power_state_list.append(sheet.cell(row,col_sec+1).value.strip())
            supply_power_list.append(sheet.cell(row,col_sec+2).value.strip())
            supply_ground_list.append(sheet.cell(row,col_sec+3).value.strip())
        supply_set_list = list(filter(None,supply_set_list))
        power_state_list = list(filter(None,power_state_list))
        supply_power_list = list(filter(None,supply_power_list))
        supply_ground_list = list(filter(None,supply_ground_list))
        for row in range(row_init,row_sec):
            for col in range(col_init,col_last):
                matrix[row-row_init][col-col_init] = sheet.cell(row,col).value.strip()
        matrix_power_state_group = [sublist for sublist in matrix if any(item != '' for item in sublist)]
        col_len = len(matrix_power_state_group[0])
        row_len = len(matrix_power_state_group)
        check_spare_matrix = True
        for row in range(1,row_len):            
            if(len(list(filter(None,matrix_power_state_group[row-1]))) != len(list(filter(None,matrix_power_state_group[row])))):
                check_spare_matrix = False
                break
        check_spare_list = (len(supply_set_list) == len(power_state_list)) and (len(supply_power_list) == len(supply_ground_list)) and (len(supply_power_list) == len(power_state_list))
        if((check_spare_list != True) or (check_spare_matrix != True)) :
            print("found empty value in <new_pst> sheet, <supply-set> , please fill in")
        else :
           #begin power_state table add
            power_state_block = power_state_block + POWER_STATE_TABLE_HEAD
            create_pst = ''
            for i in range(len(supply_power_list)):
                if('FULL_ON' in supply_power_list[i]):
                    #empty voltage
                    if(re.findall(r'\d+\.\d+',supply_power_list[i]) == []):
                        print("not found voltage value in <new_pst> sheet <"+supply_set_list[i]+", <supply-power>>, you should add voltage value like that: <FULL_ON 0.75v>")
                    else:
                        power_state_block = power_state_block + ADD_POWER_STATE_ON %(supply_set_list[i],supply_set_list[i].split('_',1)[1].strip(),re.findall(r'\d+\.\d+',supply_power_list[i])[0])
                elif('OFF' in supply_power_list[i]):
                    power_state_block = power_state_block + ADD_POWER_STATE_OFF %(supply_set_list[i],supply_set_list[i].split('_',1)[1].strip())
                else :
                    print("not found <FULL_ON or OFF> in <new_pst> sheet, the location is col: <supply-power>, row:" + supply_set_list[i])

            power_state_block = power_state_block + ADD_POWER_STATE_GROUND

            temp_power_state_list = [''] * (row_len) 
            for num in range(1,len(matrix_power_state_group[0])):
                create_pst = create_pst+ matrix_power_state_group[0][num]+'.power'+' '
            power_state_block = power_state_block + CREATE_PST %('GRP_SOC_PST',create_pst)
            for row in range(1,row_len):
                for col in range(1,col_len):
                    temp_power_state_list[row] = temp_power_state_list[row] + ' ' + matrix_power_state_group[0][col].split('_',1)[1].strip() + '_' + matrix_power_state_group[row][col]
                power_state_block = power_state_block + ADD_PST_STATE %(matrix_power_state_group[row][0],'GRP_SOC_PST',temp_power_state_list[row].strip())
    else :
        print("not found <supply-set> or <power-state-group> in <new_pst> sheet,please check your excel")

    return power_state_block
